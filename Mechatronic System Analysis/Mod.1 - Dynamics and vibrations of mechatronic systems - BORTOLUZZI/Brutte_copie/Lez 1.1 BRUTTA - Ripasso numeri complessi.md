$$
\begin{matrix}
j = \sqrt{-1} \\
s = a+j \cdot b \\
a = Ra(s) \ b = Im(s)\\
\alpha = ang (s) = arctan(b/a)\\
|s|=m=\sqrt{a^2+b^2}\\
s1+s2 = (a1+j b1)(a2+j b2) = a1 a2 - b1 b2 + j (a1 b2 + a2 b1)\\
| s1 s2 | = |s1| | s2| = m1 m2
Ang(s1 s2)= Ang (s1) + ang(s2)= \alpha1+\alpha 2\\
\end{matrix}
$$








## Coniugato
$$
\begin{matrix}
s^* = a-j b\\
s s^* = a^2 + b^2 = |s|^2\\
s + s^* = a + jb + a-jb + a -jb = 2a\\
s - s^* = a + jb -a + jb = 2jb
\end{matrix}
$$

## Notazione esponenziale
$$
\begin{matrix}
e^{j \alpha} = cos(\alpha) + j sin(\alpha)\\
s = m e^{j\alpha}\\
e^s = e^{a+jb}= e^a e^{jb}=e^a(cos b + j sin b)\\
cos(\alpha) = \frac{e^{j \alpha}+e^{-j\alpha}}{2}\\
sin(\alpha) = \frac{e^{j \alpha}-e^{-j \alpha}}{2}
\end{matrix}
$$
